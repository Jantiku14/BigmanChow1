    function formatPrice(val) {
      return '₦' + val.toFixed(2);
    }

    function saveCart() {
      localStorage.setItem('cart', JSON.stringify(cart));
      updateCartUI();
    }

    function renderPrice(product) {
      if (product.priceHigher) {
        return `<span>${formatPrice(product.priceLower)}</span> / <span>${formatPrice(product.priceHigher)}</span>`;
      } else {
        return `<span>${formatPrice(product.priceLower)}</span>`;
      }
    }

    function renderProducts(filter = '') {
      const f = filter.trim().toLowerCase();
      productList.innerHTML = '';
      const filtered = PRODUCTS.filter(p => p.title.toLowerCase().includes(f) || p.description.toLowerCase().includes(f));
      if (filtered.length === 0) {
        productList.innerHTML = '<p>No products found.</p>';
        return;
      }
      filtered.forEach(product => {
        const card = document.createElement('article');
        card.tabIndex = 0;
        card.className = 'product-card';
        card.setAttribute('role', 'listitem');
        let highlightTitle = product.title;
        if (f.length > 0) {
          const re = new RegExp(`(${f.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')})`, 'gi');
          highlightTitle = product.title.replace(re, '<span class="highlighted">$1</span>');
        }
        card.innerHTML = `
          <img src="${product.image}" alt="Image of product: ${product.title}" loading="lazy" />
          <div class="product-title">${highlightTitle}</div>
          <div class="product-prices">${renderPrice(product)}</div>
          <div class="product-description">${product.description.startsWith('*') ? product.description.replace(/\*/g, '') : product.description}</div>
        `;
        card.addEventListener('click', () => openPriceSelector(product));
        productList.appendChild(card);
      });
    }

    function openPriceSelector(product) {
      if (product.priceHigher) {
        showModal(product);
      } else {
        addToCart(product, product.priceLower, null);
      }
    }

    function showModal(product) {
      currentModalProduct = product;
      priceOptions.innerHTML = '';

      const opt1 = document.createElement('button');
      opt1.className = 'price-option';
      opt1.tabIndex = 0;
      opt1.type = 'button';
      opt1.innerHTML = `<span>Regular Size - ${formatPrice(product.priceLower)}</span>`;
      opt1.addEventListener('click', () => {
        addToCart(product, product.priceLower, 'Regular Size');
        closeModal();
      });

      const opt2 = document.createElement('button');
      opt2.className = 'price-option';
      opt2.tabIndex = 0;
      opt2.type = 'button';
      opt2.innerHTML = `<span>Large Size - ${formatPrice(product.priceHigher)}</span>`;
      opt2.addEventListener('click', () => {
        addToCart(product, product.priceHigher, 'Large Size');
        closeModal();
      });

      priceOptions.appendChild(opt1);
      priceOptions.appendChild(opt2);
      priceModal.classList.add('show');
      priceModal.setAttribute('aria-hidden', 'false');
      priceModal.focus();
      document.body.style.overflow = 'hidden';
    }
    function closeModal() {
      priceModal.classList.remove('show');
      priceModal.setAttribute('aria-hidden', 'true');
      document.body.style.overflow = '';
      currentModalProduct = null;
    }

    let cart = JSON.parse(localStorage.getItem('cart')) || [];

    function addToCart(product, selectedPrice, priceLabel) {
      const existing = cart.find(i => i.id === product.id && i.selectedPrice === selectedPrice);
      if (existing) {
        existing.quantity++;
      } else {
        cart.push({ id: product.id, quantity: 1, selectedPrice, selectedLabel: priceLabel });
      }
      saveCart();
      showToast(`Added ${product.title}${priceLabel ? ' (' + priceLabel + ')' : ''} to cart`, 'check_circle');
      updateCartUI();
    }

    function removeFromCart(productId, selectedPrice) {
      cart = cart.filter(i => !(i.id === productId && i.selectedPrice === selectedPrice));
      saveCart();
      showToast('Removed item from cart', 'delete');
      updateCartUI();
    }

    function updateQuantity(productId, selectedPrice, delta) {
      const item = cart.find(i => i.id === productId && i.selectedPrice === selectedPrice);
      if (!item) return;
      item.quantity += delta;
      if (item.quantity < 1) {
        removeFromCart(productId, selectedPrice);
        return;
      }
      saveCart();
      updateCartUI();
    }

    function updateCartUI() {
      cartItems.innerHTML = '';
      if (cart.length === 0) {
        cartItems.innerHTML = '<p>Your cart is empty.</p>';
        btnSubmit.disabled = true;
        cartBadge.textContent = '0';
        return;
      }
      btnSubmit.disabled = false;
      cartBadge.textContent = cart.reduce((acc, i) => acc + i.quantity, 0);
      let total = 0;
      cart.forEach(item => {
        const p = PRODUCTS.find(prod => prod.id === item.id);
        if (!p) return;
        const price = item.selectedPrice;
        total += price * item.quantity;
        const div = document.createElement('div');
        div.className = 'cart-item';
        div.tabIndex = 0;
        div.setAttribute('role', 'listitem');
        const priceLabelText = item.selectedLabel ? ` (${item.selectedLabel})` : '';
        div.innerHTML = `
          <img src="${p.image}" alt="Image of product in cart: ${p.title}" class="cart-thumb" loading="lazy" />
          <div class="cart-info">
            <div class="cart-title">${p.title}${priceLabelText}</div>
            <div class="cart-price">${formatPrice(price * item.quantity)}</div>
          </div>
          <div class="cart-qty-controls" aria-label="Quantity controls for ${p.title}${priceLabelText}">
            <button class="cart-qty-btn" aria-label="Decrease quantity of ${p.title}${priceLabelText}">&minus;</button>
            <span aria-live="polite" aria-atomic="true">${item.quantity}</span>
            <button class="cart-qty-btn" aria-label="Increase quantity of ${p.title}${priceLabelText}">&plus;</button>
          </div>
        `;
        const btnMinus = div.querySelectorAll('.cart-qty-btn')[0];
        const btnPlus = div.querySelectorAll('.cart-qty-btn')[1];
        btnMinus.addEventListener('click', () => updateQuantity(item.id, item.selectedPrice, -1));
        btnPlus.addEventListener('click', () => updateQuantity(item.id, item.selectedPrice, 1));
        cartItems.appendChild(div);
      });
      cartTotalPrice.textContent = formatPrice(total);
    }

    function showToast(message, icon) {
      const toast = document.createElement('div');
      toast.className = 'toast';
      toast.setAttribute('role', 'alert');
      toast.setAttribute('aria-live', 'assertive');
      toast.style.background = 'var(--color-primary)';
      toast.style.color = '#fff';
      toast.style.padding = '12px 24px';
      toast.style.borderRadius = '16px';
      toast.style.boxShadow = '0 4px 12px rgba(0,0,0,0.3)';
      toast.style.marginTop = '12px';
      toast.style.display = 'flex';
      toast.style.alignItems = 'center';
      toast.style.gap = '12px';
      toast.style.fontWeight = '600';
      const iconSpan = document.createElement('span');
      iconSpan.className = 'material-icons';
      iconSpan.textContent = icon || 'info';
      iconSpan.setAttribute('aria-hidden', 'true');
      toast.appendChild(iconSpan);
      toast.appendChild(document.createTextNode(message));
      TOAST_CONTAINER.appendChild(toast);
      setTimeout(() => {
        toast.style.opacity = '0';
        setTimeout(() => TOAST_CONTAINER.removeChild(toast), 350);
      }, 3000);
    }
    function validateField(field) {
      const value = field.value.trim();
      const id = field.id;
      let error = '';
      switch (id) {
        case 'fullname':
          if (!value) error = 'Full name is required';
          else if (value.length < 3) error = 'Full name must be at least 3 characters';
          break;
        case 'address':
          if (!value) error = 'Delivery address is required';
          break;
        case 'city':
          if (!value) error = 'City is required';
          break;
        case 'payment':
          if (!value) error = 'Please select a payment method';
          break;
      }
      const errorId = id + '-error';
      const errorDiv = document.getElementById(errorId);
      if (error) {
        errorDiv.textContent = error;
        field.setAttribute('aria-invalid', 'true');
        return false;
      } else {
        errorDiv.textContent = '';
        field.removeAttribute('aria-invalid');
        return true;
      }
    }
    function validateForm() {
      let valid = true;
      const fields = [...checkoutForm.querySelectorAll('input, select')];
      fields.forEach(field => {
        const f = validateField(field);
        if (!f) valid = false;
      });
      return valid;
    }
    function updateSubmitState() {
      btnSubmit.disabled = !(validateForm() && cart.length > 0);
    }
    checkoutForm.addEventListener('input', e => {
      if (e.target.matches('input, select')) {
        validateField(e.target);
        updateSubmitState();
      }
    });
    checkoutForm.addEventListener('submit', e => {
      e.preventDefault();
      const paymentMethod = document.getElementById('payment').value;
      if (paymentMethod === 'bank-transfer') {
        // Redirect to bank transfer info page
        window.location.href = 'bank-transfer.html';
        return;
      }
      if (!validateForm()) {
        showToast('Please fix errors before submitting', 'error');
        return;
      }
      if (cart.length === 0) {
        showToast('Your cart is empty', 'warning');
        return;
      }
      btnSubmit.disabled = true;
      showToast('Processing your order...', 'info');
      setTimeout(() => {
        showToast('Order placed successfully!', 'check_circle');
        cart = [];
        saveCart();
        checkoutForm.reset();
        updateSubmitState();
        document.activeElement.blur();
      }, 2000);
    });
    function setTheme(theme) {
      if (theme === 'dark') {
        body.setAttribute('data-theme', 'dark');
        themeToggle.innerHTML = '<span class="material-icons">light_mode</span>';
        localStorage.setItem('theme', 'dark');
      } else {
        body.setAttribute('data-theme', 'light');
        themeToggle.innerHTML = '<span class="material-icons">dark_mode</span>';
        localStorage.setItem('theme', 'light');
      }
    }
    themeToggle.addEventListener('click', () => {
      const current = body.getAttribute('data-theme');
      setTheme(current === 'light' ? 'dark' : 'light');
    });
    const savedTheme = localStorage.getItem('theme');
    if (savedTheme) {
      setTheme(savedTheme);
    } else {
      const prefersDark = window.matchMedia('(prefers-color-scheme: dark)').matches;
      setTheme(prefersDark ? 'dark' : 'light');
    }
    function openSidebar() {
      sidebar.classList.add('mobile-open');
      sidebarOverlay.classList.add('show');
      mobileMenuToggle.setAttribute('aria-pressed', 'true');
      trapFocus(sidebar);
    }
    function closeSidebar() {
      sidebar.classList.remove('mobile-open');
      sidebarOverlay.classList.remove('show');
      mobileMenuToggle.setAttribute('aria-pressed', 'false');
      releaseFocusTrap();
    }
    mobileMenuToggle.addEventListener('click', () => {
      if (sidebar.classList.contains('mobile-open')) {
        closeSidebar();
      } else {
        openSidebar();
      }
    });
    sidebarOverlay.addEventListener('click', closeSidebar);
    let focusableElems = [];
    let firstFocusable = null;
    let lastFocusable = null;
    let focusTrapActive = false;
    function trapFocus(element) {
      focusableElems = element.querySelectorAll('a, button, input, select, textarea, [tabindex]:not([tabindex="-1"])');
      if (focusableElems.length === 0) return;
      firstFocusable = focusableElems[0];
      lastFocusable = focusableElems[focusableElems.length - 1];
      focusTrapActive = true;
      firstFocusable.focus();
      document.addEventListener('keydown', handleTrap);
    }
    function releaseFocusTrap() {
      focusTrapActive = false;
      document.removeEventListener('keydown', handleTrap);
      mobileMenuToggle.focus();
    }
    function handleTrap(e) {
      if (!focusTrapActive) return;
      if (e.key === 'Tab') {
        if (e.shiftKey) {
          if (document.activeElement === firstFocusable) {
            e.preventDefault();
            lastFocusable.focus();
          }
        } else {
          if (document.activeElement === lastFocusable) {
            e.preventDefault();
            firstFocusable.focus();
          }
        }
      } else if (e.key === 'Escape') {
        closeSidebar();
      }
    }
    document.addEventListener('keydown', (e) => {
      if (e.target.tagName === 'INPUT' || e.target.tagName === 'TEXTAREA' || e.target.isContentEditable) return;
      if (e.key === 'a' || e.key === 'A') {
        if (PRODUCTS.length > 0) {
          addToCart(PRODUCTS[0], PRODUCTS[0].priceLower, null);
          showToast('Added first product to cart via shortcut "A"', 'check_circle');
        }
      }
    });
    let searchTimeout = null;
    searchInput.addEventListener('input', () => {
      clearTimeout(searchTimeout);
      searchTimeout = setTimeout(() => {
        renderProducts(searchInput.value);
      }, 250);
    });

    renderProducts();
    updateCartUI();
    updateSubmitState();

  })();
